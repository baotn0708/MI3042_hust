\section{Giải bài tập}

\begin{bt}
Tính định thức Vandermonde trong định lý về sự tồn tại duy nhất nghiệm của đa thức nội suy.
\end{bt}

\begin{loigiai}
Định thức Vandermonde cấp $n$ ứng với các số $x_1, x_2, \dots, x_n$ có dạng:
\[
    V_n = \begin{vmatrix}
        1 & x_1 & x_1^2 & \cdots & x_1^{n-1} \\
        1 & x_2 & x_2^2 & \cdots & x_2^{n-1} \\
        \vdots & \vdots & \vdots & \ddots & \vdots \\
        1 & x_n & x_n^2 & \cdots & x_n^{n-1}
    \end{vmatrix}.
\]
Khi $n=2$, ta có:
\[
V_2 = \begin{vmatrix}
    1 & x_1 \\
    1 & x_2
\end{vmatrix} = x_2 - x_1.
\]
Khi $n=3$, ta có:
\[
\begin{aligned}
        V_3 = \begin{vmatrix}
        1 & x_1 & x_1^2 \\
        1 & x_2 & x_2^2 \\
        1 & x_3 & x_3^2
    \end{vmatrix} &= 1 (x_2 x_3^2 - x_3 x_2^2) - 1 (x_1 x_3^2 - x_3 x_1^2) + 1 (x_1 x_2^2 - x_2 x_1^2) \\
    &= x_2^2(-x_3+x_1) + x_2(x_3^2-x_1^2)  +(-x_1 x_3^2 + x_3 x_1^2) \\
    &= (x_1-x_3)(x_2^2-x_2(x_1+x_3) + x_1 x_3) \\
    &= (x_1-x_3)(x_2-x_1)(x_2-x_3).
\end{aligned}
\]
Giả sử đẳng thức $V_n = \prod_{1 \leq i < j \leq n} (x_j - x_i)$ đúng tới cho $n = k-1$. Ta sẽ chứng minh rằng đẳng thức này cũng đúng cho $n = k$.

\[
\begin{aligned}
    V_k &= \begin{vmatrix}
    1 & x_1 & x_1^2 & \cdots & x_1^{k-1} \\
    1 & x_2 & x_2^2 & \cdots & x_2^{k-1} \\
    \vdots & \vdots & \vdots & \ddots & \vdots \\
    1 & x_k & x_k^2 & \cdots & x_k^{k-1}
\end{vmatrix} \\&= \begin{vmatrix}
    1 & x_1 & x_1^2 & \cdots & x_1^{k-1}\\
    0 & x_2-x_1 & x_2^2-x_1^2 & \cdots & x_2^{k-1}-x_1^{k-1} \\
    \vdots & \vdots & \vdots & \ddots & \vdots \\
    0 & x_k-x_1 & x_k^2-x_1^2 & \cdots & x_k^{k-1}-x_1^{k-1}
\end{vmatrix} \\ 
&= \begin{vmatrix}
    x_2-x_1 & x_2^2-x_1^2 & \cdots & x_2^{k-1}-x_1^{k-1} \\
    x_3-x_1 & x_3^2-x_1^2 & \cdots & x_3^{k-1}-x_1^{k-1} \\
    \vdots & \vdots & \vdots & \ddots & \vdots \\
    x_k-x_1 & x_k^2-x_1^2 & \cdots & x_k^{k-1}-x_1^{k-1}
\end{vmatrix} \\
& = (x_2-x_1)(x_3-x_1)\cdots(x_k-x_1)\begin{vmatrix}
    1 & x_2+x_1 & \cdots & \sum_{j=0}^{k-2} x_2^j x_1^{k-2-j} \\
    1 & x_3+x_1 & \cdots & \sum_{j=0}^{k-2} x_3^j x_1^{k-2-j} \\
    \vdots & \vdots & \ddots & \vdots \\
    1 & x_k+x_1 & \cdots & \sum_{j=0}^{k-2} x_k^j x_1^{k-2-j}
\end{vmatrix}
\end{aligned}
\]

\end{loigiai}
